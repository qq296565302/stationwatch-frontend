### 二、自动记录时间的实现机制

#### 前端实现（JavaScript/Vue）

```javascript
// 自动时间记录逻辑
data() {
  return {
    dutyItems: [],
    currentItem: {
      content: '',
      startTime: null,
      endTime: null,
      isCompleted: false
    }
  }
},

methods: {
  // 监听内容输入，自动记录开始时间
  onContentInput(item) {
    if (item.content && !item.startTime) {
      item.startTime = this.formatTime(new Date());
    }
  },
  
  // 点击"标记完成"，自动记录结束时间
  markAsCompleted(item) {
    if (!item.endTime) {
      item.endTime = this.formatTime(new Date());
      item.isCompleted = true;
    }
  },
  
  // 时间格式化为 HH:MM
  formatTime(date) {
    return date.getHours().toString().padStart(2, '0') + ':' + 
           date.getMinutes().toString().padStart(2, '0');
  },
  
  // 添加新记录
  addNewItem() {
    this.dutyItems.push({
      id: Date.now(),
      content: '',
      startTime: null,
      endTime: null,
      isCompleted: false
    });
  }
}
```

#### 后端验证（Python/FastAPI）

```python
from datetime import datetime
from pydantic import BaseModel, validator
from typing import List, Optional

class DutyItem(BaseModel):
    content: str
    start_time: Optional[str] = None
    end_time: Optional[str] = None
    is_completed: bool = False
    
    @validator('start_time', 'end_time')
    def validate_time_format(cls, v):
        if v:
            try:
                datetime.strptime(v, '%H:%M')
            except ValueError:
                raise ValueError('时间格式必须为HH:MM')
        return v

class DutyRecord(BaseModel):
    record_date: str
    weather: str
    duty_items: List[DutyItem]
    other_matters: str
    pending_issues: str

# API端点
@app.post("/api/records")
async def create_duty_record(record: DutyRecord, current_user: User = Depends(get_current_user)):
    # 验证时间合法性
    for item in record.duty_items:
        if item.end_time and item.start_time:
            # 确保结束时间大于开始时间
            start = datetime.strptime(item.start_time, '%H:%M')
            end = datetime.strptime(item.end_time, '%H:%M')
            if end < start:
                raise HTTPException(status_code=400, detail="结束时间必须大于开始时间")
    
    # 保存到数据库
    db_record = DutyRecordModel(
        record_date=record.record_date,
        weather=record.weather,
        creator_id=current_user.id,
        # ... 其他字段
    )
    db.add(db_record)
    
    # 保存明细项
    for item in record.duty_items:
        db_item = DutyItemModel(
            record_id=db_record.id,
            content=item.content,
            start_time=item.start_time,
            end_time=item.end_time,
            is_completed=item.is_completed
        )
        db.add(db_item)
    
    db.commit()
    return {"message": "值班记录已保存", "id": db_record.id}
```

### 三、Excel导出功能实现

这是最关键的部分，需要保持原有Excel模板的格式。

#### 数据库查询数据

```python
from sqlalchemy.orm import Session
from models import DutyRecord, DutyItem
from datetime import datetime

def get_monthly_records(db: Session, year: int, month: int):
    """查询指定月份的所有值班记录"""
    start_date = datetime(year, month, 1)
    if month == 12:
        end_date = datetime(year + 1, 1, 1)
    else:
        end_date = datetime(year, month + 1, 1)
    
    records = db.query(DutyRecord).filter(
        DutyRecord.record_date >= start_date,
        DutyRecord.record_date < end_date
    ).all()
    
    return records
```

#### Excel导出逻辑

```python
from openpyxl import load_workbook
from openpyxl.styles import Font, Alignment, Border, Side
import os

def export_monthly_excel(records: list, year: int, month: int, template_path: str):
    """
    导出月度值班记录Excel文件
    保持原有模板格式，将Web数据填充到Excel
    """
    
    # 加载模板
    template_file = os.path.join(template_path, '电子值班表模板.xlsx')
    wb = load_workbook(template_file)
    
    # 创建新文件
    output_file = f'值班记录_{year}-{str(month).zfill(2)}.xlsx'
    
    for record in records:
        # 为每条记录创建一个worksheet
        sheet_name = f"{year}-{str(month).zfill(2)}-{record.record_date.day}"
        ws = wb.create_sheet(title=sheet_name)
        
        # 复制模板结构（标题、表头等）
        copy_template_structure(wb['Sheet1'], ws)
        
        # 填充标题
        ws['B2'] = f"{record.station_name}{year}年{month}月份值班记录"
        
        # 填充天气情况
        ws['B4'] = f"天气情况：{record.weather}"
        
        # 填充值班事项（行6-16，动态填充）
        row_num = 6
        for item in record.duty_items[:11]:  # 限制最多11条，保持表格美观
            ws[f'C{row_num}'] = item.content
            ws[f'B{row_num}'] = item.start_time if item.start_time else ''
            ws[f'D{row_num}'] = item.end_time if item.end_time else ''
            row_num += 1
        
        # 如果记录少于11条，清空剩余行
        for i in range(row_num, 17):
            ws[f'A{i}'] = '时'
            ws[f'I{i}'] = '时'
            ws[f'B{i}'] = ''
            ws[f'J{i}'] = ''
        
        # 填充其他事项（行17-19）
        ws.merge_cells('A17:K17')
        ws['A17'] = f"其他事项：{record.other_matters}"
        
        # 填充遗留问题（行20-21）
        ws.merge_cells('A20:K21')
        ws['A20'] = f"遗留问题：{record.pending_issues}"
    
    # 删除模板sheet
    if 'Sheet1' in wb.sheetnames:
        del wb['Sheet1']
    
    # 保存文件
    wb.save(output_file)
    return output_file

def copy_template_structure(source_ws, target_ws):
    """复制模板的样式和结构"""
    # 复制列宽
    for col in source_ws.columns:
        target_ws.column_dimensions[col[0].column_letter].width = source_ws.column_dimensions[col[0].column_letter].width
    
    # 复制行高
    for row_num in range(1, 30):
        target_ws.row_dimensions[row_num].height = source_ws.row_dimensions[row_num].height
    
    # 复制单元格样式（标题、表头等）
    for row in source_ws.iter_rows(min_row=1, max_row=5):
        for cell in row:
            new_cell = target_ws[cell.coordinate]
            new_cell.font = Font(name=cell.font.name, size=cell.font.size, bold=cell.font.bold)
            new_cell.alignment = cell.alignment
    
    # 复制边框
    thin_border = Border(
        left=Side(style='thin'),
        right=Side(style='thin'),
        top=Side(style='thin'),
        bottom=Side(style='thin')
    )
    
    # 设置数据区域边框（A6:K16）
    for row in range(6, 17):
        for col in ['A', 'B', 'C', 'D', 'E', 'F', 'G', 'H', 'I', 'J', 'K']:
            target_ws[f'{col}{row}'].border = thin_border
```

```python
from fastapi import FastAPI, Depends, HTTPException
from fastapi.middleware.cors import CORSMiddleware
from fastapi.responses import FileResponse
from sqlalchemy.orm import Session
import os

app = FastAPI()

# CORS配置（允许前端访问）
app.add_middleware(
    CORSMiddleware,
    allow_origins=["http://localhost:3000"],  # 前端地址
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

# 认证相关
@app.post("/api/auth/login")
async def login(credentials: LoginRequest, db: Session = Depends(get_db)):
    # 验证用户名密码
    user = authenticate_user(db, credentials.username, credentials.password)
    if not user:
        raise HTTPException(status_code=401, detail="用户名或密码错误")
    
    # 生成JWT token
    access_token = create_access_token(data={"sub": user.username})
    return {"access_token": access_token, "user": user.to_dict()}

# 值班记录相关
@app.post("/api/records")
async def create_record(
    record: DutyRecordCreate,
    current_user: User = Depends(get_current_user),
    db: Session = Depends(get_db)
):
    """创建新的值班记录"""
    # 验证权限（只有值班员及以上角色可以创建）
    if current_user.role not in ['duty_officer', 'supervisor', 'admin']:
        raise HTTPException(status_code=403, detail="无权限创建记录")
    
    # 验证日期（只能创建当日或未来日期的记录）
    if datetime.strptime(record.record_date, '%Y-%m-%d').date() < datetime.now().date():
        raise HTTPException(status_code=400, detail="不能创建历史日期的记录")
    
    # 保存到数据库
    db_record = save_duty_record(db, record, current_user.id)
    return {"message": "记录创建成功", "id": db_record.id}

@app.get("/api/records/{record_id}")
async def get_record(
    record_id: int,
    current_user: User = Depends(get_current_user),
    db: Session = Depends(get_db)
):
    """获取单条值班记录"""
    record = db.query(DutyRecord).filter(DutyRecord.id == record_id).first()
    if not record:
        raise HTTPException(status_code=404, detail="记录不存在")
    
    # 权限检查：值班员只能查看自己的记录
    if current_user.role == 'duty_officer' and record.creator_id != current_user.id:
        raise HTTPException(status_code=403, detail="无权限查看此记录")
    
    return record.to_dict()

@app.put("/api/records/{record_id}")
async def update_record(
    record_id: int,
    record_update: DutyRecordUpdate,
    current_user: User = Depends(get_current_user),
    db: Session = Depends(get_db)
):
    """更新值班记录"""
    db_record = db.query(DutyRecord).filter(DutyRecord.id == record_id).first()
    if not db_record:
        raise HTTPException(status_code=404, detail="记录不存在")
    
    # 权限检查
    if current_user.role == 'duty_officer':
        if db_record.creator_id != current_user.id:
            raise HTTPException(status_code=403, detail="无权限修改此记录")
        if db_record.record_date < datetime.now().date():
            raise HTTPException(status_code=403, detail="不能修改历史记录")
    
    # 更新记录
    update_duty_record(db, db_record, record_update)
    return {"message": "记录更新成功"}

# 导出功能
@app.get("/api/records/export/monthly/{year}/{month}")
async def export_monthly_records(
    year: int,
    month: int,
    current_user: User = Depends(get_current_user),
    db: Session = Depends(get_db)
):
    """导出月度值班记录为Excel"""
    # 权限检查
    if current_user.role not in ['supervisor', 'admin']:
        raise HTTPException(status_code=403, detail="无权限导出记录")
    
    # 查询数据
    records = get_monthly_records(db, year, month)
    if not records:
        raise HTTPException(status_code=404, detail="该月份无值班记录")
    
    # 生成Excel文件
    template_path = os.path.join(os.getcwd(), 'templates')
    excel_file = export_monthly_excel(records, year, month, template_path)
    
    # 返回文件下载
    return FileResponse(
        excel_file,
        media_type='application/vnd.openxmlformats-officedocument.spreadsheetml.sheet',
        filename=f'值班记录_{year}-{str(month).zfill(2)}.xlsx'
    )
```

```python
# models.py

from sqlalchemy import create_engine, Column, Integer, String, DateTime, Boolean, Text, ForeignKey
from sqlalchemy.ext.declarative import declarative_base
from sqlalchemy.orm import relationship
from datetime import datetime

Base = declarative_base()

class User(Base):
    __tablename__ = 'users'
    
    id = Column(Integer, primary_key=True)
    username = Column(String(50), unique=True, nullable=False)
    password_hash = Column(String(255), nullable=False)
    real_name = Column(String(50), nullable=False)
    role_id = Column(Integer, ForeignKey('roles.id'))
    department = Column(String(100))
    is_active = Column(Boolean, default=True)
    created_at = Column(DateTime, default=datetime.now)
    last_login = Column(DateTime)
    
    role = relationship("Role", back_populates="users")
    duty_records = relationship("DutyRecord", back_populates="creator")

class Role(Base):
    __tablename__ = 'roles'
    
    id = Column(Integer, primary_key=True)
    role_name = Column(String(50), unique=True, nullable=False)
    permissions = Column(Text)  # JSON格式存储权限列表
    description = Column(String(200))
    
    users = relationship("User", back_populates="role")

class DutyRecord(Base):
    __tablename__ = 'duty_records'
    
    id = Column(Integer, primary_key=True)
    record_date = Column(DateTime, nullable=False)
    station_name = Column(String(100), nullable=False)
    weather = Column(String(20))
    creator_id = Column(Integer, ForeignKey('users.id'))
    is_locked = Column(Boolean, default=False)
    created_at = Column(DateTime, default=datetime.now)
    updated_at = Column(DateTime, default=datetime.now, onupdate=datetime.now)
    
    creator = relationship("User", back_populates="duty_records")
    duty_items = relationship("DutyItem", back_populates="record", cascade="all, delete-orphan")

class DutyItem(Base):
    __tablename__ = 'duty_items'
    
    id = Column(Integer, primary_key=True)
    record_id = Column(Integer, ForeignKey('duty_records.id'), nullable=False)
    content = Column(Text)
    start_time = Column(String(5))  # 格式：HH:MM
    end_time = Column(String(5))
    is_completed = Column(Boolean, default=False)
    created_at = Column(DateTime, default=datetime.now)
    
    record = relationship("DutyRecord", back_populates="duty_items")

class OtherMatter(Base):
    __tablename__ = 'other_matters'
    
    id = Column(Integer, primary_key=True)
    record_id = Column(Integer, ForeignKey('duty_records.id'), nullable=False)
    content = Column(Text)
    created_at = Column(DateTime, default=datetime.now)

class PendingIssue(Base):
    __tablename__ = 'pending_issues'
    
    id = Column(Integer, primary_key=True)
    record_id = Column(Integer, ForeignKey('duty_records.id'), nullable=False)
    content = Column(Text)
    is_resolved = Column(Boolean, default=False)
    created_at = Column(DateTime, default=datetime.now)

class SystemConfig(Base):
    __tablename__ = 'system_config'
    
    id = Column(Integer, primary_key=True)
    config_key = Column(String(50), unique=True, nullable=False)
    config_value = Column(String(200))
    description = Column(String(200))

class OperationLog(Base):
    __tablename__ = 'operation_logs'
    
    id = Column(Integer, primary_key=True)
    user_id = Column(Integer, ForeignKey('users.id'))
    operation_type = Column(String(50))
    target_table = Column(String(50))
    target_id = Column(Integer)
    ip_address = Column(String(50))
    created_at = Column(DateTime, default=datetime.now)
```
